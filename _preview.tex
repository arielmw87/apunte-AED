% Muestra un diagrama. Argumento: "capX/nombre" (e.g. cap1/latch-SR).
% Si existe tikz/capX/ref/nombre.png (captura del original de Floyd),
% lo pone AL LADO de la versión TikZ para comparar.
% Si no existe ref, muestra solo el TikZ.
\newcommand{\showtikz}[1]{%
  % Extraer capítulo y nombre base
  \StrCount{#1}{/}[\slashcount]%
  \ifnum\slashcount>0
    \StrBefore{#1}{/}[\tikzchap]%
    \StrBehind{#1}{/}[\tikzbase]%
    \edef\refpath{tikz/\tikzchap/ref/\tikzbase.png}%
  \else
    \def\tikzchap{}%
    \def\tikzbase{#1}%
    \edef\refpath{tikz/ref/#1.png}%
  \fi
  \begin{center}
  {\sffamily\bfseries\large\texttt{#1.tikz}}\par\vspace{6mm}
  \IfFileExists{\refpath}{%
    \begin{minipage}[t]{0.47\textwidth}\centering
      {\footnotesize\itshape Original (Floyd)}\par\vspace{3mm}
      \includegraphics[width=\linewidth,keepaspectratio]{\refpath}
    \end{minipage}\hfill
    \begin{minipage}[t]{0.47\textwidth}\centering
      {\footnotesize\itshape Versión TikZ}\par\vspace{3mm}
      \resizebox{\linewidth}{!}{\input{tikz/#1.tikz}}
    \end{minipage}%
  }{%
    \IfFileExists{tikz/#1.tikz}{\input{tikz/#1.tikz}}{\textit{(pendiente)}}%
  }%
  \end{center}\newpage}
